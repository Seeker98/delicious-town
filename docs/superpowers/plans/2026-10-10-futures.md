## Review Focus

1. 两家同时订同一种、额度只剩 1 份：只有一家成功，另一家整单回滚（定金不扣）。→ Task 3 测试。
2. 参考价跨过 0 点变了，页面上的旧单价下单：报 `futures_price_moved` 带新价，不扣钱。→ Task 3 测试。

```ts
import type { Tuning } from '@dt/config';

export type FuturesTuning = Tuning['futures'];

/** 期货单价（设计 §3）：基准 = clamp(参考价, 等级价, 等级价 × capRate)，单价 = ceil(基准 × premium) */
export function futuresUnitPrice(levelPrice: number, ref: number, t: FuturesTuning): number {
  const base = Math.min(Math.max(ref, levelPrice), levelPrice * t.capRate);
  return Math.ceil(base * t.premium);
}

/** 定金 = ceil(总价 × depositRate)；尾款 = 总价 − 定金 */
export const futuresDeposit = (total: number, t: FuturesTuning): number => Math.ceil(total * t.depositRate);

/** 区服每天每种的额度：列表里单独设了的优先，否则按等级默认 */
export const futuresQuota = (level: number, override: number | null, t: FuturesTuning): number =>
  override ?? t.dailyQuota[level - 1] ?? 0;
```

注意浮点：`9000 × 1.2 = 10800.000000000002` 时 `ceil` 会变 10801。实现时用 `Math.ceil(Math.round(base * t.premium * 1e6) / 1e6)` 消掉误差，定金同理；测试第一条覆盖。

- [ ] **Step 4: 实现服务**（要点，代码按此写全）：
  - `order(ctx, b)`：先 `ensureFeature(ctx.shardId, 'exchange')`，再“暖”参考价（照 `exchange/service.ts` 的 `warmRef`：锁外用 `d.db` 调一次 `refPrice`），然后 `runOp(d, ctx, { feature: 'futures', source: 'futures' }, (o) => ...)`：
    1. 冻结 → `invalidState('exchange_frozen')`；门槛 → `requirement`（同交易所 place 的三行）。
    2. 读 `futures_food` 这一行；`food = o.config.foods.get(id)`；`!food || food.retired || !row?.enabled || food.level < 1 || food.level > 5` → `futures_not_listed`。
    3. `day = gameDay(o.now)`；`ref = refPrice(o.tx, o.config, o.tuning.exchange, o.tuning.market.levelPriceRate, o.shardId, id, day)`；`unit = futuresUnitPrice(foodPrice(food, o.tuning.market), ref, t)`；`unit !== b.unitPrice` → `futures_price_moved { price: unit }`。
    4. 个人：`used = getDaily(o.tx, rest.id, 'futures.qty', day)`；`used + qty > personDaily` → `limitReached('futures_person', { max, left: max(0, personDaily − used) })`。

       ```ts
       const quota = futuresQuota(food.level, row.daily_quota, t);
       const got = await sql<{ used: number }>`
         insert into futures_quota (shard_id, foods_id, day, used)
         select ${o.shardId}, ${id}, ${day}, ${b.qty} where ${b.qty} <= ${quota}
         on conflict (shard_id, foods_id, day) do update set used = futures_quota.used + excluded.used
           where futures_quota.used + excluded.used <= ${quota}
         returning used`.execute(o.tx);
       if (got.rows.length === 0) {
         const cur = await o.tx.selectFrom('futures_quota').select('used')
           .where('shard_id', '=', o.shardId).where('foods_id', '=', id).where('day', '=', day).executeTakeFirst();
         throw limitReached('futures_quota', { max: quota, left: Math.max(0, quota - (cur?.used ?? 0)) });
       }
       ```
    7. `incrementDaily(o.tx, rest.id, 'futures.qty', qty, day)`；插入 `futures_contract`（`due_at = now + deliverHours`）；`restLog(o, 'futures.order', { foodsId, qty, unitPrice, deposit, balance })`；返回 DTO。
  - `cancel(ctx, id)`：`runOp(..., { feature: 'restaurant', source: 'futures' })`；`update futures_contract set status='cancelled', settled_at=now where id and rest_id=me and status='open' and due_at > now returning *`；没有行 → `futures_not_open`；`restLog(o, 'futures.cancel', { foodsId, qty, deposit })`。

```ts
export async function deliverDue(d: GameDeps, shardId: number, now: Date, log: Log = NO_LOG) {
  const due = await d.db.selectFrom('futures_contract').select(['id', 'rest_id'])
    .where('shard_id', '=', shardId).where('status', '=', 'open').where('due_at', '<=', now)
    .orderBy('rest_id').orderBy('due_at').orderBy('id').limit(500).execute();
  const byRest = new Map<number, string[]>();
  for (const r of due) byRest.set(r.rest_id, [...(byRest.get(r.rest_id) ?? []), r.id]);
  const stats = { delivered: 0, defaulted: 0, refunded: 0, failed: 0 };
  for (const [restId, ids] of byRest) {
    try {
      await runSystemOp(d, shardId, restId, { source: 'futures', now }, async (op) => {
        const rows = await op.tx.selectFrom('futures_contract').selectAll()
          .where('id', 'in', ids).where('status', '=', 'open').orderBy('due_at').orderBy('id').forUpdate().execute();
        const credits = newCredits();
        for (const r of rows) {
          const set = (s: Partial<...>) => op.tx.updateTable('futures_contract').set({ settled_at: now, ...s }).where('id', '=', r.id).execute();
          if (!op.config.foods.get(r.foods_id)) {
            gainCoin(op, Number(r.deposit), { source: 'futures.refund' });
            await set({ status: 'cancelled' });
            restLog(op, 'futures.refunded', { foodsId: r.foods_id, qty: r.qty, deposit: Number(r.deposit) });
            stats.refunded++; continue;
          }
          if (op.rest.coin >= Number(r.balance)) {
            spendCoin(op, Number(r.balance), { source: 'futures.balance' });
            const plan = await addFoods(op, r.foods_id, r.qty, { source: 'futures', keepDropped: true });
            if (plan.dropped > 0) addCredit(credits, op.rest.id, 0, r.foods_id, plan.dropped);
            await set({ status: 'delivered', to_cupboard: r.qty - plan.dropped, to_wallet: plan.dropped });
            restLog(op, 'futures.delivered', { foodsId: r.foods_id, qty: r.qty, toWallet: plan.dropped });
            stats.delivered++;
          } else {
            await set({ status: 'defaulted' });
            restLog(op, 'futures.defaulted', { foodsId: r.foods_id, qty: r.qty, deposit: Number(r.deposit) });
            stats.defaulted++;
          }
        }
        await creditWallets(op.tx, credits);
      });
    } catch (err) {
      log.error({ err, shardId, restId }, 'futures deliver failed');
      stats.failed++;
    }
  }
  return stats;
}
```
