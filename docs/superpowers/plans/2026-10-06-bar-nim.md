> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 酒吧加第 8 个游戏“最后一颗糖”：两张桌子，玩家和调酒师轮流拿 1~k 颗糖，拿到最后一颗的赢。

**Architecture:** 照 4C-3 的记忆调酒、飞镖做：局面存在 `bar_round`（`game = 'nim'`），每日次数用 `daily_counter`（`bar.nim`），奖励走酒吧共用的 `randomAward`，声望用 `gainRenown`。调酒师的拿法是纯函数，放在 `rules.ts`，单独测试。前端加一个 `NimPanel`。

**Tech Stack:** Fastify + Kysely + Postgres；Vue 3 + Pinia；zod；vitest。

## Review Focus

1. 调酒师拿的数量永远在 1~min(k, 剩余) 之内，剩 1 颗时也不会出错。
2. 高手桌抛到调酒师先拿、而调酒师第一步就拿完。这不会发生，因为糖果数下限大于 k 上限；数值校验要保证这一点，区服覆盖也一样。
3. 并发：同一局连点两次“拿”。操作已经锁了店；第二次读到的是新局面，越界就报 `badInput`，局已结束就报 `no_round`。
4. 刷新后接着下：`/bar` 返回的 `nim.round` 和 `NimDto` 一样，前端能直接接上，包括新手桌还没选先后的状态。
5. 礼券不够、次数用完时不扣礼券、不计次数、不建局。

### Task 3: 调酒师的拿法（纯函数）

**Files:**
- Modify: `apps/server/src/modules/bar/rules.ts`
- Test: `apps/server/src/modules/bar/rules.test.ts`

**Interfaces:**
- Produces: `export function nimBartenderTake(left: number, k: number, mistake: number, rng: Rng): number`

- [ ] **Step 1**：写失败的测试。用 `sequenceRng`，也就是 `@dt/shared` 里那个按给定序列吐 `next()` 的随机数：
  - **占优、不失手**：`(10, 3, 0, rng)` 返回 2（10 mod 4）；`(7, 3, 0)` 返回 3。
  - **占优、新手桌**：
    - `mistake 0.5`，`next()` 依次是 0.6 → 不失手，返回 r。
    - 0.4 → 失手，再用 `int(min(k, left))` 抽，`next()` 是 0 时返回 1。
  - **必输局面**：`(8, 3, 0)` 时 `next()` 是 0.99，返回 3，也就是 `1 + int(3)`。
  - **剩 1 颗**：返回 1。
  - **剩 2 颗、k=3**：返回 2，正好是 r。
  - **随机遍历**：任意 `left` 在 1~40、`k` 在 3~5、`mistake` 取 0 / 0.5 / 1，跑 1000 次，返回值都在 `[1, min(k, left)]`。
- [ ] **Step 2**：`pnpm -F @dt/server test -- src/modules/bar/rules.test.ts`。Expected：FAIL（函数不存在）。
- [ ] **Step 3**：实现

```ts
/**
 * 最后一颗糖的调酒师（设计 §4.2）：剩余 mod (k+1) 不为 0 时拿余数，让剩下的是 k+1 的倍数；
 * 新手桌有 mistake 的概率失手；必输局面在 1~min(k, 剩余) 里随便拿
 */
export function nimBartenderTake(left: number, k: number, mistake: number, rng: Rng): number {
  const r = left % (k + 1);
  if (r !== 0 && !rng.chance(mistake)) return r;
  return 1 + rng.int(Math.min(k, left));
}
```

  注意：`rng.chance(0)` 是 `next() < 0`，永远 false，但仍会消耗一个 `next()`。测试里的序列要按这个排。

```ts
import { GOODS } from '@dt/config';
import { ErrorCode, gameDay, type NimDto, type NimTable } from '@dt/shared';
import { emitAction } from '../../core/action';
import { invalidState, limitReached } from '../../core/errors';
import { restLog, type Op } from '../../core/op';
import { gainRenown } from '../../core/resources';
import { randomAward } from '../award/random';
import { getDaily, incrementDaily } from '../counter/dailyCounter';
import { consumeGoods } from '../store/goods';
import { badInput } from './common';
import { assertNoRound, endRound, loadRound, saveRound } from './round';
import { nimBartenderTake } from './rules';

/** 服务端局面（设计 §5）；全部公开，没有要藏的 */
export interface NimState {
  table: NimTable;
  k: number;
  pile: number;
  left: number;
  /** 新手桌还没选先后时为 false */
  started: boolean;
  coin: 'me' | 'bartender' | null;
  log: Array<{ who: 'me' | 'bartender'; take: number }>;
}

const between = (o: Op, [lo, hi]: readonly [number, number]) => lo + o.rng.int(hi - lo + 1);

export function nimView(s: NimState): NimDto {
  return {
    table: s.table, k: s.k, pile: s.pile, left: s.left, log: s.log,
    needFirst: !s.started, coin: s.coin, result: null, renown: 0, award: null,
  };
}

/** 调酒师拿一步，拿完返回 true */
function bartender(o: Op, s: NimState): boolean {
  const take = nimBartenderTake(s.left, s.k, o.tuning.bar.nim.tables[s.table].mistake, o.rng);
  s.left -= take;
  s.log.push({ who: 'bartender', take });
  return s.left === 0;
}

/** 结束：赢了发声望和奖励；删掉局面、写个人日志 */
async function finish(o: Op, s: NimState, win: boolean): Promise<NimDto> {
  const t = o.tuning.bar.nim.tables[s.table];
  await endRound(o, 'nim');
  let award = null;
  if (win) {
    gainRenown(o, t.renown);
    award = await randomAward(o, { level: t.awardLevel, noTicket: true, bar: true });
  }
  restLog(o, 'bar.nim', { table: s.table, result: win ? 'win' : 'lose' });
  return { ...nimView(s), result: win ? 'win' : 'lose', renown: win ? t.renown : 0, award };
}

/** 开局（设计 §4.3）：扣礼券、计次数；高手桌抛硬币，调酒师先拿时开局就有他一步 */
export async function nimStart(o: Op, table: NimTable): Promise<NimDto> {
  const n = o.tuning.bar.nim;
  const t = n.tables[table];
  await assertNoRound(o, 'nim');
  const day = gameDay(o.now);
  if ((await getDaily(o.tx, o.rest.id, 'bar.nim', day)) >= n.dailyMax)
    throw limitReached('bar_daily', { max: n.dailyMax });
  await consumeGoods(o, GOODS.mysteryTicket, t.cost);
  await incrementDaily(o.tx, o.rest.id, 'bar.nim', 1, day);
  const k = between(o, t.k);
  const pile = between(o, t.pile);
  const s: NimState = { table, k, pile, left: pile, started: t.first !== 'choose', coin: null, log: [] };
  if (t.first === 'coin') {
    s.coin = o.rng.chance(0.5) ? 'me' : 'bartender';
    // 糖果数下限大于 k 上限（数值校验），调酒师第一步拿不完
    if (s.coin === 'bartender') bartender(o, s);
  }
  await saveRound(o, 'nim', s);
  await emitAction(o, 'bar.play');
  await emitAction(o, 'bar.nim');
  return nimView(s);
}

/** 新手桌选先后 */
export async function nimFirst(o: Op, who: 'me' | 'bartender'): Promise<NimDto> {
  const s = await loadRound<NimState>(o, 'nim');
  if (!s) throw invalidState('no_round');
  if (s.started) throw invalidState('started');
  s.started = true;
  if (who === 'bartender') bartender(o, s);
  await saveRound(o, 'nim', s);
  return nimView(s);
}

/** 玩家拿 num 颗；没拿完时调酒师接着拿 */
export async function nimTake(o: Op, num: number): Promise<NimDto> {
  const s = await loadRound<NimState>(o, 'nim');
  if (!s) throw invalidState('no_round');
  if (!s.started) throw invalidState('need_first');
  if (num < 1 || num > s.k || num > s.left) throw badInput('num');
  s.left -= num;
  s.log.push({ who: 'me', take: num });
  if (s.left === 0) return finish(o, s, true);
  if (bartender(o, s)) return finish(o, s, false);
  await saveRound(o, 'nim', s);
  return nimView(s);
}
```
